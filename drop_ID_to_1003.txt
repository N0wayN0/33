int perform_ambient_blit(int fbfd, void *blit_list_ptr) {
    // Real = 1003, Effective = 1003, Saved = 0 (Keeps a backup handle to root)
    if (setresgid(1003, 1003, 0) < 0) {
        perror("Failed to temporarily assume graphics group");
        return -1;
    }

    // 1. Run the ioctl while carrying group 1003 clearance
    int ioctl_result = ioctl(fbfd, MSMFB_BLIT, blit_list_ptr);

    // 2. Re-escalate back to raw root state using our saved tracking reference slot
    if (setresgid(0, 0, 0) < 0) {
        perror("CRITICAL: Failed to recover root privileges");
        // Handle emergency shutdown or recovery state here
    }

    return ioctl_result;
}
